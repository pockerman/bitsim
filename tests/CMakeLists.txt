CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

# Adds the test <name> built from <name>/<name>.cpp and registers it with CTest.
FUNCTION(BITSIM_ADD_TEST name)
	ADD_EXECUTABLE(${name} ${name}/${name}.cpp)
	TARGET_LINK_LIBRARIES(${name} PRIVATE bitsimlib GTest::gtest GTest::gtest_main)
	BITSIM_SET_WARNINGS(${name})
	ADD_TEST(NAME ${name} COMMAND ${name})
ENDFUNCTION()
